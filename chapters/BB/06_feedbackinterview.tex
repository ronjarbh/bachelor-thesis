\textbf{HF-4 -- Akzeptanztreiber und -barrieren}

\textbf{Struktur, Orientierung und Auffindbarkeit}
\begin{itemize}
  \item Klare Strukturen und schnelle Orientierung als zentrale Akzeptanztreiber 
  (TN-03, TN-05, TN-06, TN-09).
  \item Eindeutige Verlinkungen sowie logisch aufgebaute Übersichts- und Playbook-Seiten fördern den direkten Informationszugang 
  (TN-04).
  \item Fehlende \acrshort{bzw} schwache Suchfunktion sowie geringere Auffindbarkeit im Vergleich zu anderen Quellen wirken hemmend 
  (TN-01, TN-09).
\end{itemize}

\textbf{Konkreter Arbeitsalltagsnutzen und Use-Case-Bezug}
\begin{itemize}
  \item Konkreter Nutzen für den Arbeitsalltag sowie Unterstützung typischer Use Cases erhöhen die Akzeptanz 
  (TN-01, TN-04).
  \item Wenn der Nutzen nicht unmittelbar erkennbar ist, wirkt dies hemmend auf die Nutzung 
  (TN-01, TN-06).
\end{itemize}

\textbf{Zentrale Wissensquelle und Informationsweitergabe}
\begin{itemize}
  \item Möglichkeit, Kolleginnen und Kollegen einfacher zu informieren und Wissen weiterzugeben 
  (TN-01, TN-10).
  \item Leicht weiterleitbare Dokumentationen und ein zentraler Wissenspunkt als Zielbild 
  (TN-03, TN-09).
  \item Geringe Akzeptanz im Team oder in angrenzenden Bereichen kann die Nutzung hemmen 
  (TN-09).
\end{itemize}

\textbf{Content-Qualität: Aktualität, Vollständigkeit und Informationsmenge}
\begin{itemize}
  \item Hohe Aktualität der Inhalte wird als zentraler Akzeptanztreiber wahrgenommen 
  (TN-06, TN-07).
  \item Veraltete oder unvollständige Inhalte wirken hemmend 
  (TN-03, TN-04, TN-05).
  \item Zu viele Informationen auf einmal („Overload“) erschweren die Nutzung 
  (TN-05).
\end{itemize}

\textbf{Einführung, Kommunikation und Erwartungsklarheit}
\begin{itemize}
  \item Regelmäßige Hinweise in Meetings sowie klare Einführung und Onboarding fördern die Nutzung 
  (TN-02).
  \item Wiederkehrende sichtbare Impulse (\acrshort{zB} Tipps oder Reminder) können die Akzeptanz weiter steigern 
  (TN-06).
  \item Unklare Erwartungen zur Nutzung sowie parallele Wissensquellen mit ähnlichen Inhalten wirken hemmend 
  (TN-01, TN-09).
\end{itemize}
