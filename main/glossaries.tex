% GLOBAL GLOSSARY



% ------------------------------------------------
% GLOSSARIES

\newglossaryentry{uxdes}{
    name={User Experience},
    description={Gesamtheit der Wahrnehmungen, Reaktionen und Bewertungen, die eine Person vor, während und nach der Nutzung eines Systems erlebt}
}

\newglossaryentry{infarchitektur}{
    name={Informationsarchitektur},
    description={Disziplin zur Strukturierung, Benennung und Verknüpfung von Inhalten, um Navigation, Auffindbarkeit und Verständnis innerhalb eines Informationssystems zu unterstützen}
}

\newglossaryentry{uxstrategien}{
    name={UX-Strategien},
    description={Übergeordnete Leitlinien und Ziele zur systematischen Gestaltung konsistenter, nutzerzentrierter
            Produkt- und Serviceerlebnisse}
}

\newglossaryentry{mentalemodelle}{
    name={mentale Modelle},
    description={Kognitive Repräsentationen, anhand derer Nutzer Erwartungen über Funktionsweisen eines Systems bilden und Handlungen planen}
}

\newglossaryentry{affordanz}{
    name={Affordanzen},
    description={Gestaltungsmerkmale, die Nutzern mögliche Handlungen mit einem Objekt oder einer Oberfläche signalisieren}
}

\newglossaryentry{feedback}{
    name={Feedback},
    description={Vom System bereitgestellte Informationen über den aktuellen Zustand oder das Ergebnis einer Nutzeraktion}
}

\newglossaryentry{cognitiveload}{
    name={kognitive Belastung},
    description={Ausmaß der mentalen Anstrengung, die zur Verarbeitung von Informationen und Ausführung von Aufgaben erforderlich ist}
}

\newglossaryentry{singlesourceoftruth}{
    name={Single Source of Truth},
    description={Prinzip, nach dem eine zentrale, autoritative Informationsquelle gepflegt wird, um Redundanzen und Inkonsistenzen zu vermeiden}
}

\newglossaryentry{informationssilos}{
    name={Informationssilos},
    description={Strukturell oder organisatorisch getrennte Wissensbereiche, die den Austausch und die Zugänglichkeit von Informationen einschränken}
}

\newglossaryentry{metadaten}{
    name={Metadaten},
    description={Daten, die Inhalte beschreiben, klassifizieren oder strukturieren und dadurch deren Auffindbarkeit und Einordnung erleichtern}
}

\newglossaryentry{klassifikationssysteme}{
    name={Klassifikationssysteme},
    description={Strukturierte Schemata zur Ordnung von Inhalten, z.\,B. nach Themen, Aufgaben oder Nutzergruppen}
}

\newglossaryentry{navigationsstruktur}{
    name={Navigationsstruktur},
    description={Logische und visuelle Organisation der Wege, über die Nutzer Inhalte in einem System erreichen können}
}

\newglossaryentry{workaround}{
    name={Workarounds},
    description={Umgehungslösungen, die genutzt werden, wenn ein System benötigte Funktionen nicht angemessen bereitstellt}
}

\newglossaryentry{wissensmanagement}{
    name={Wissensmanagement},
    description={Organisierter Prozess der Erfassung, Strukturierung und Bereitstellung von Wissen innerhalb einer Organisation}
}

\newglossaryentry{digitaleTransformation}{
    name={digitale Transformation},
    description={Tiefgreifender organisatorischer Wandel, der sich aus der Einführung, Integration und Weiterentwicklung digitaler Technologien ergibt}
}

\newglossaryentry{hybridarbeit}{
    name={hybride Arbeitsumgebungen},
    description={Arbeitsmodelle, die physische und digitale Arbeitsorte sowie verteilte Teams kombinieren}
}

\newglossaryentry{humancentereddesign}{
    name={Human-Centered Design},
    description={Gestaltungsansatz nach ISO 9241-210, der auf einer systematischen Analyse von Nutzern, Aufgaben und Nutzungskontexten basiert und iterative Evaluationen einfordert}
}

\newglossaryentry{enterpriseux}{
    name={Enterprise UX},
    description={User-Experience-Gestaltung für komplexe, organisationsinterne Systeme mit Fokus auf Prozessunterstützung, Effizienz und Konsistenz}
}

\newglossaryentry{interaktionsdesign}{
    name={Interaktionsdesign},
    description={Gestaltung der Abläufe, Zustände und Rückmeldungen, die das Verhalten eines Systems im Dialog mit dem Nutzer bestimmen}
}

\newglossaryentry{governance}{
    name={Content-Governance},
    description={Regelungen, Verantwortlichkeiten und Prozesse, die die Qualität, Pflege und Aktualität von Inhalten in Informationssystemen sicherstellen}
}

\newglossaryentry{medienbruch}{
    name={Medienbrüche},
    description={Unterbrechungen eines Arbeitsprozesses, die entstehen, wenn zwischen getrennten Systemen oder Kanälen gewechselt werden muss}
}

\newglossaryentry{usability}{
    name={Usability},
    description={Ausmaß, in dem ein Produkt zur Zielerreichung effizient, effektiv und zufriedenstellend genutzt werden kann}
}

\newglossaryentry{iterativesdesign}{
    name={Iteratives Design},
    description={Vorgehensweise, bei der Gestaltungslösungen schrittweise entwickelt, getestet und anhand von Feedback optimiert werden}
}

\newglossaryentry{rollenmodelle}{
    name={Rollenmodelle},
    description={Strukturierte Beschreibung von Zuständigkeiten, Aufgaben und Interaktionsbeziehungen innerhalb einer Organisation}
}

\newglossaryentry{playbook}{
    name={Playbook},
    description={Sammlung standardisierter Vorgehensweisen, Leitlinien oder Best Practices zur Durchführung wiederkehrender Aufgaben oder Prozesse}
}

\newglossaryentry{wikispace}{
    name={Wiki-Space},
    description={Abgegrenzter Bereich innerhalb eines Wiki-Systems, der zusammenhängende Inhalte, Navigation und Berechtigungen umfasst}
}

\newglossaryentry{enterprisesystem}{
    name={Enterprise-System},
    description={Unternehmensweites Softwaresystem, das Arbeitsprozesse, Wissensmanagement und Kommunikation unterstützt und skaliert}
}

\newglossaryentry{tamdes}{
    name={Technology Acceptance Model (TAM)},
    description={Ein von Davis (1986) entwickeltes Modell zur Erklärung der Akzeptanz von Informationssystemen.
            Es geht davon aus, dass die Nutzungsabsicht primär durch die \gls{pu}
            und die \gls{peou} beeinflusst wird}
}

\newglossaryentry{utautdes}{
    name={Unified Theory of Acceptance and Use of Technology (UTAUT)},
    description={Ein von Venkatesh et al.\ (2003) entwickeltes Modell, das verschiedene Akzeptanzmodelle integriert.
            Es identifiziert Leistungserwartung, Aufwandserwartung, sozialer Einfluss und
            unterstützende Rahmenbedingungen als zentrale Einflussfaktoren auf die Nutzungsintention
            und das tatsächliche Nutzungsverhalten}
}

\newglossaryentry{pu}{
    name={wahrgenommene Nützlichkeit (Perceived Usefulness, PU)},
    description={Eine zentrale Dimension des \Gls{tam}, die beschreibt, in welchem Ausmaß eine Person glaubt,
            dass die Nutzung eines Systems die eigene Arbeitsleistung verbessert}
}

\newglossaryentry{peou}{
    name={wahrgenommene Benutzerfreundlichkeit (Perceived Ease of Use, PEOU)},
    description={Ein Kernelement des \Gls{tam}, das beschreibt, wie einfach und aufwandsarm ein System subjektiv zu bedienen ist}
}

\newglossaryentry{performanceexpectancy}{
    name={Leistungserwartung (Performance Expectancy)},
    description={Ein Konstrukt des \Gls{utaut}-Modells, das ausdrückt, inwieweit Nutzer davon ausgehen,
            dass die Verwendung einer Technologie ihre Aufgabenleistung verbessert}
}

\newglossaryentry{effortexpectancy}{
    name={Aufwandserwartung (Effort Expectancy)},
    description={Ein Konstrukt des \Gls{utaut}-Modells, das beschreibt, wie leicht und intuitiv die Bedienung einer Technologie wahrgenommen wird}
}

\newglossaryentry{personas}{
    name={Personas},
    description={Fiktive, aber datengestützte Repräsentationen typischer Nutzergruppen.
            Personas bündeln Ziele, Bedürfnisse, Verhaltensmuster und Kontextinformationen, um
            nutzerzentrierte Designentscheidungen zu unterstützen}
}

\newglossaryentry{userjourneys}{
    name={User Journeys},
    description={Darstellungen der einzelnen Schritte oder Interaktionspunkte, die Nutzer auf dem Weg zur Erreichung eines bestimmten Ziels durchlaufen.
            Sie machen Bedürfnisse, Probleme und Emotionen im Nutzungsprozess sichtbar}
}

\newglossaryentry{szenarien}{
    name={Szenarien},
    description={Narrative Beschreibungen typischer Nutzungssituationen, die Kontext, Ziele und Aufgaben von Nutzern im Umgang mit einer Technologie konkretisieren.
            Sie dienen dazu, Designanforderungen realitätsnah abzuleiten}
}

\newglossaryentry{nutzungserwartungen}{
    name={Nutzungserwartungen},
    description={Vorstellungen und Annahmen von Nutzern darüber, welchen Nutzen, welche Funktionsweisen und welche
            Bedienqualität eine digitale Plattform bieten sollte}
}

\newglossaryentry{musterkataloge}{
    name={Musterkataloge (Pattern Libraries)},
    description={Sammlungen wiederkehrender, bewährter Lösungsansätze für Interface- oder Interaktionsprobleme.
            Sie unterstützen Konsistenz, Effizienz und die systematische Gestaltung digitaler Produkte}
}

\newglossaryentry{agil}{
    name={Agile Arbeitsumfelder},
    description={Organisationale Rahmenbedingungen, die iterative, inkrementelle und nutzerzentrierte Vorgehensweisen begünstigen.
            Typisch sind kurze Entwicklungszyklen, flexible Anpassungen und eine enge Zusammenarbeit interdisziplinärer Teams}
}

\newglossaryentry{rqdes}{
    name={Forschungsfragen},
    description={Präzise formulierte Fragen, die den Untersuchungsfokus einer wissenschaftlichen Arbeit bestimmen.
            Sie leiten methodische Entscheidungen und strukturieren die Analyse- und Entwicklungsprozesse}
}

\newglossaryentry{poc}{
    name={Proof-of-Concept},
    description={Machbarkeitsnachweis, der demonstriert, dass ein Konzept oder eine Idee praktisch umsetzbar ist}
}

\newglossaryentry{dsr}{
    name={Design-Science-Research},
    description={Forschungsansatz, der sich auf die Entwicklung und Evaluation innovativer Artefakte konzentriert,
            um praktische Probleme zu lösen und theoretisches Wissen zu erweitern}
}

\newglossaryentry{uxwriting}{
    name={UX Writing},
    description={Gestaltung funktionaler Texte in digitalen Interfaces, die Nutzende bei der Interaktion unterstützen, Orientierung bieten und Unsicherheiten reduzieren und eng mit Informationsarchitektur und Interaktionslogik verzahnt sind}
}

\newglossaryentry{microinteractions}{
    name={Microinteractions},
    description={Kleine, in sich geschlossene Interaktionsabläufe mit Auslöser, Regeln und Feedback, die Systemzustände sichtbar machen, Handlungen bestätigen und das Erleben von Kontrolle und Verständlichkeit unterstützen}
}

\newglossaryentry{transformationsdesign}{
    name={Transformationsdesign},
    description={Bewusste Gestaltung organisationaler Veränderungsprozesse unter Einbeziehung nutzerzentrierter Prinzipien, bei der UX-Methoden mit Change-Management-Ansätzen verknüpft werden, um Veränderungen nachvollziehbar und erlebbar zu machen}
}

\newglossaryentry{changecommunication}{
    name={Change Communication},
    description={Geplante Kommunikationsmaßnahmen zur Begleitung von Veränderungsprozessen, die Orientierung schaffen, Erwartungen steuern und den Sinn neuer Systeme und Arbeitsweisen vermitteln}
}

\newglossaryentry{change}{
    name={Change},
    description={Organisatorischer Wandel, der durch neue Technologien, Prozesse oder Strukturen ausgelöst wird und Anpassungen in Arbeitsweisen, Rollen und Unternehmenskultur erfordert}
}

\newglossaryentry{designpatterns}{
    name={Design Patterns},
    description={Wiederkehrende, bewährte Lösungsmuster für spezifische Designprobleme, die Konsistenz und Effizienz in der Gestaltung digitaler Produkte fördern}
}

\newglossaryentry{leanux}{
    name={Lean-UX},
    description={Ein agiler, nutzerzentrierter Ansatz im UX-Design, der schnelle Iterationen, kontinuierliches Feedback und enge Zusammenarbeit im Team betont, um effiziente und effektive Nutzererlebnisse zu schaffen}
}
